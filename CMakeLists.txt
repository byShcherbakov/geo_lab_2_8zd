cmake_minimum_required(VERSION 3.16)
project(CompGeomGraphLab2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
add_executable(CompGeomGraphLab2 Main.cpp func.cpp Model.cpp tasks.cpp Demo.cpp Ui.cpp Experiment.cpp)
target_link_libraries(CompGeomGraphLab2 PRIVATE OpenGL::GL GLEW::GLEW glfw glm::glm)
target_compile_definitions(CompGeomGraphLab2 PRIVATE GLEW_NO_GLU GLFW_INCLUDE_NONE LAB_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(CompGeomGraphLab2 PRIVATE /utf-8)
endif()
foreach(shader vertex_shader.txt fragment_shader.txt line_geometry.txt)
 add_custom_command(TARGET CompGeomGraphLab2 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${shader}" "$<TARGET_FILE_DIR:CompGeomGraphLab2>/${shader}")
endforeach()
